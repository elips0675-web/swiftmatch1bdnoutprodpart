## 📄 Артефакт 2: `scripts/run-tests.sh`

```bash
#!/bin/bash
# scripts/run-tests.sh — единый прогон всех тестов SwiftMatch
# Использование: bash scripts/run-tests.sh [--json] [--quiet]
set -e

# Цвета
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

# Флаги
JSON_OUTPUT=false
QUIET=false
for arg in "$@"; do
  case $arg in
    --json) JSON_OUTPUT=true ;;
    --quiet) QUIET=true ;;
  esac
done

# Временные файлы для JSON-отчётов
TMP_DIR=$(mktemp -d)
FRONT_JSON="$TMP_DIR/front.json"
SERVER_JSON="$TMP_DIR/server.json"

cleanup() {
  rm -rf "$TMP_DIR"
}
trap cleanup EXIT

log() {
  if [ "$QUIET" = false ]; then
    echo -e "${BLUE}[$(date +%H:%M:%S)]${NC} $1"
  fi
}

fail() {
  echo -e "${RED}✗ $1${NC}"
  exit 1
}

success() {
  echo -e "${GREEN}✓ $1${NC}"
}

warn() {
  echo -e "${YELLOW}⚠ $1${NC}"
}

# ===== 1. Frontend tests =====
log "Frontend tests (Vitest)..."
if [ "$JSON_OUTPUT" = true ]; then
  npm run test -- --reporter=json > "$FRONT_JSON" 2>/dev/null || fail "Frontend tests failed"
else
  npm run test || fail "Frontend tests failed"
fi

# ===== 2. Server tests =====
log "Server tests (Vitest)..."
cd server
if [ "$JSON_OUTPUT" = true ]; then
  npm run test -- --reporter=json > "$SERVER_JSON" 2>/dev/null || fail "Server tests failed"
else
  npm run test || fail "Server tests failed"
fi
cd ..

# ===== 3. Summary =====
if [ "$JSON_OUTPUT" = true ]; then
  node -e "
    const f = require('$FRONT_JSON');
    const s = require('$SERVER_JSON');
    const total = f.numTotalTests + s.numTotalTests;
    const passed = f.numPassedTests + s.numPassedTests;
    const failed = f.numFailedTests + s.numFailedTests;
    console.log(JSON.stringify({
      frontend: { total: f.numTotalTests, passed: f.numPassedTests, failed: f.numFailedTests },
      server: { total: s.numTotalTests, passed: s.numPassedTests, failed: s.numFailedTests },
      total: { total, passed, failed },
      status: failed === 0 ? 'PASS' : 'FAIL'
    }, null, 2));
    process.exit(failed > 0 ? 1 : 0);
  "
else
  echo ""
  echo -e "${BLUE}═══════════════════════════════════════${NC}"
  echo -e "${BLUE}         TEST SUMMARY                  ${NC}"
  echo -e "${BLUE}═══════════════════════════════════════${NC}"

  # Парсим числа из JSON-отчётов
  FRONT_TOTAL=$(node -e "console.log(require('$FRONT_JSON').numTotalTests)")
  FRONT_PASSED=$(node -e "console.log(require('$FRONT_JSON').numPassedTests)")
  SERVER_TOTAL=$(node -e "console.log(require('$SERVER_JSON').numTotalTests)")
  SERVER_PASSED=$(node -e "console.log(require('$SERVER_JSON').numPassedTests)")

  TOTAL=$((FRONT_TOTAL + SERVER_TOTAL))
  PASSED=$((FRONT_PASSED + SERVER_PASSED))

  if [ "$FRONT_PASSED" -eq "$FRONT_TOTAL" ]; then
    success "Frontend: $FRONT_PASSED/$FRONT_TOTAL tests passed"
  else
    fail "Frontend: $FRONT_PASSED/$FRONT_TOTAL tests passed"
  fi

  if [ "$SERVER_PASSED" -eq "$SERVER_TOTAL" ]; then
    success "Server:   $SERVER_PASSED/$SERVER_TOTAL tests passed"
  else
    fail "Server: $SERVER_PASSED/$SERVER_TOTAL tests passed"
  fi

  echo -e "${BLUE}───────────────────────────────────────${NC}"
  if [ "$PASSED" -eq "$TOTAL" ]; then
    success "TOTAL:    $PASSED/$TOTAL tests passed ✓"
  else
    fail "TOTAL: $PASSED/$TOTAL tests passed ✗"
  fi
  echo -e "${BLUE}═══════════════════════════════════════${NC}"
fi
```

**Сделать исполняемым:**
```bash
chmod +x scripts/run-tests.sh
```

### Артефакт 5: `scripts/coverage-gap.mjs`

```javascript
#!/usr/bin/env node
/**
 * scripts/coverage-gap.mjs — поиск непокрытых модулей
 * 
 * Использование:
 *   node scripts/coverage-gap.mjs              # все модули
 *   node scripts/coverage-gap.mjs --server     # только сервер
 *   node scripts/coverage-gap.mjs --frontend   # только фронт
 *   node scripts/coverage-gap.mjs --json       # JSON-вывод
 * 
 * Как работает:
 * 1. Сканирует все .js/.ts файлы в server/src/ и src/
 * 2. Исключает __tests__/, test/, *.test.*, *.spec.*
 * 3. Проверяет, есть ли соответствующий тестовый файл
 * 4. Выводит список модулей без тестов
 */

import { readdirSync, statSync, existsSync } from 'fs';
import { join, dirname, basename } from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);
const root = join(__dirname, '..');

// Флаги
const args = process.argv.slice(2);
const SERVER_ONLY = args.includes('--server');
const FRONTEND_ONLY = args.includes('--frontend');
const JSON_OUTPUT = args.includes('--json');

// Паттерны для исключения
const EXCLUDE_PATTERNS = [
  '__tests__',
  'test',
  'tests',
  '.test.',
  '.spec.',
  'node_modules',
  'dist',
  'build',
  'coverage',
  '.git',
  'android',
  'ios',
  'e2e',
  'scripts',
  'database',
  'backups',
  'nginx',
  'docker',
  '.github'
];

// Маппинг: исходный файл → ожидаемый тестовый файл
function getExpectedTestFile(sourceFile, isServer) {
  const relativePath = sourceFile.replace(root + '/', '');
  const parts = relativePath.split('/');
  
  if (isServer) {
    // server/src/routes/auth.js → server/src/__tests__/auth.test.js
    const fileName = basename(sourceFile, '.js');
    return join(root, 'server/src/__tests__', `${fileName}.test.js`);
  } else {
    // src/components/Button.tsx → src/test/button.test.tsx
    const fileName = basename(sourceFile, '.tsx').toLowerCase();
    return join(root, 'src/test', `${fileName}.test.tsx`);
  }
}

// Рекурсивный обход директории
function walkDir(dir, files = []) {
  if (!existsSync(dir)) return files;
  
  const entries = readdirSync(dir);
  for (const entry of entries) {
    const fullPath = join(dir, entry);
    const stat = statSync(fullPath);
    
    if (stat.isDirectory()) {
      if (!EXCLUDE_PATTERNS.some(p => entry.includes(p))) {
        walkDir(fullPath, files);
      }
    } else if (stat.isFile()) {
      if (/\.(js|ts|tsx)$/.test(entry) && !EXCLUDE_PATTERNS.some(p => entry.includes(p))) {
        files.push(fullPath);
      }
    }
  }
  return files;
}

// Основная логика
function findCoverageGaps() {
  const gaps = { server: [], frontend: [] };
  
  // Сервер
  if (!FRONTEND_ONLY) {
    const serverFiles = walkDir(join(root, 'server/src'));
    for (const file of serverFiles) {
      // Пропускаем утилиты, конфиги, middleware (они часто покрыты интеграционно)
      if (file.includes('/middleware/') || file.includes('/utils/') || file.includes('/config/')) {
        continue;
      }
      
      const testFile = getExpectedTestFile(file, true);
      if (!existsSync(testFile)) {
        gaps.server.push({
          source: file.replace(root + '/', ''),
          expectedTest: testFile.replace(root + '/', ''),
          exists: false
        });
      }
    }
  }
  
  // Фронтенд
  if (!SERVER_ONLY) {
    const frontendFiles = walkDir(join(root, 'src'));
    for (const file of frontendFiles) {
      // Пропускаем утилиты, типы, константы
      if (file.includes('/lib/') || file.includes('/types/') || file.includes('/constants')) {
        continue;
      }
      
      const testFile = getExpectedTestFile(file, false);
      if (!existsSync(testFile)) {
        gaps.frontend.push({
          source: file.replace(root + '/', ''),
          expectedTest: testFile.replace(root + '/', ''),
          exists: false
        });
      }
    }
  }
  
  return gaps;
}

// Вывод
const gaps = findCoverageGaps();

if (JSON_OUTPUT) {
  console.log(JSON.stringify(gaps, null, 2));
  process.exit(gaps.server.length + gaps.frontend.length > 0 ? 1 : 0);
} else {
  console.log('\n=== COVERAGE GAPS ===\n');
  
  if (gaps.server.length > 0) {
    console.log(`🔴 Server modules without tests (${gaps.server.length}):\n`);
    for (const gap of gaps.server) {
      console.log(`  ${gap.source}`);
      console.log(`    → Expected: ${gap.expectedTest}\n`);
    }
  }
  
  if (gaps.frontend.length > 0) {
    console.log(`🔴 Frontend modules without tests (${gaps.frontend.length}):\n`);
    for (const gap of gaps.frontend) {
      console.log(`  ${gap.source}`);
      console.log(`    → Expected: ${gap.expectedTest}\n`);
    }
  }
  
  const total = gaps.server.length + gaps.frontend.length;
  if (total === 0) {
    console.log('✅ All modules have tests!\n');
    process.exit(0);
  } else {
    console.log(`\n⚠️  Total: ${total} modules without tests\n`);
    process.exit(1);
  }
}
```

### Артефакт 6: `scripts/unused-routes.mjs`

```javascript
#!/usr/bin/env node
/**
 * scripts/unused-routes.mjs — поиск роутов без тестов
 * 
 * Использование:
 *   node scripts/unused-routes.mjs              # все роуты
 *   node scripts/unused-routes.mjs --json       # JSON-вывод
 * 
 * Как работает:
 * 1. Парсит все router.get/post/put/delete в server/src/routes/
 * 2. Извлекает пути (например, '/api/users/:id')
 * 3. Проверяет, упоминается ли путь в тестах
 * 4. Выводит список роутов без тестов
 */

import { readdirSync, readFileSync, statSync, existsSync } from 'fs';
import { join, dirname, basename } from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);
const root = join(__dirname, '..');

const args = process.argv.slice(2);
const JSON_OUTPUT = args.includes('--json');

// Паттерн для извлечения роутов
const ROUTE_PATTERN = /router\.(get|post|put|delete|patch)\s*\(\s*['"`]([^'"`]+)['"`]/g;

// Рекурсивный обход директории
function walkDir(dir, files = []) {
  if (!existsSync(dir)) return files;
  
  const entries = readdirSync(dir);
  for (const entry of entries) {
    const fullPath = join(dir, entry);
    const stat = statSync(fullPath);
    
    if (stat.isDirectory()) {
      walkDir(fullPath, files);
    } else if (stat.isFile() && entry.endsWith('.js')) {
      files.push(fullPath);
    }
  }
  return files;
}

// Извлечение роутов из файла
function extractRoutes(filePath) {
  const content = readFileSync(filePath, 'utf-8');
  const routes = [];
  let match;
  
  while ((match = ROUTE_PATTERN.exec(content)) !== null) {
    const method = match[1].toUpperCase();
    const path = match[2];
    routes.push({ method, path, file: filePath });
  }
  
  return routes;
}

// Проверка, упоминается ли роут в тестах
function isRouteTested(route, testFiles) {
  const pathPattern = route.path.replace(/:\w+/g, '\\w+'); // :id → \w+
  const regex = new RegExp(pathPattern, 'i');
  
  for (const testFile of testFiles) {
    const content = readFileSync(testFile, 'utf-8');
    if (regex.test(content)) {
      return true;
    }
  }
  
  return false;
}

// Основная логика
function findUnusedRoutes() {
  const routeFiles = walkDir(join(root, 'server/src/routes'));
  const testFiles = walkDir(join(root, 'server/src/__tests__'));
  
  const allRoutes = [];
  for (const file of routeFiles) {
    const routes = extractRoutes(file);
    allRoutes.push(...routes);
  }
  
  const untested = [];
  for (const route of allRoutes) {
    if (!isRouteTested(route, testFiles)) {
      untested.push(route);
    }
  }
  
  return untested;
}

// Вывод
const untested = findUnusedRoutes();

if (JSON_OUTPUT) {
  console.log(JSON.stringify(untested, null, 2));
  process.exit(untested.length > 0 ? 1 : 0);
} else {
  console.log('\n=== ROUTES WITHOUT TESTS ===\n');
  
  if (untested.length === 0) {
    console.log('✅ All routes have tests!\n');
    process.exit(0);
  } else {
    console.log(`🔴 Found ${untested.length} routes without tests:\n`);
    
    // Группируем по файлу
    const byFile = {};
    for (const route of untested) {
      const file = basename(route.file);
      if (!byFile[file]) byFile[file] = [];
      byFile[file].push(route);
    }
    
    for (const [file, routes] of Object.entries(byFile)) {
      console.log(`\n${file}:`);
      for (const route of routes) {
        console.log(`  ${route.method} ${route.path}`);
      }
    }
    
    console.log(`\n⚠️  Total: ${untested.length} untested routes\n`);
    process.exit(1);
  }
}
```

import { vi } from 'vitest';

export function createMockRedis() {
  const store = new Map();
  
  const mockRedis = {
    get: vi.fn(async (key) => store.get(key) || null),
    set: vi.fn(async (key, value, options) => {
      store.set(key, value);
      if (options?.EX) {
        setTimeout(() => store.delete(key), options.EX * 1000);
      }
      return 'OK';
    }),
    del: vi.fn(async (key) => {
      const existed = store.has(key);
      store.delete(key);
      return existed ? 1 : 0;
    }),
    keys: vi.fn(async (pattern) => {
      const regex = new RegExp(pattern.replace('*', '.*'));
      return Array.from(store.keys()).filter(k => regex.test(k));
    }),
    expire: vi.fn(async (key, seconds) => {
      if (!store.has(key)) return 0;
      setTimeout(() => store.delete(key), seconds * 1000);
      return 1;
    }),
    incr: vi.fn(async (key) => {
      const current = parseInt(store.get(key) || '0', 10);
      const next = current + 1;
      store.set(key, String(next));
      return next;
    }),
    decr: vi.fn(async (key) => {
      const current = parseInt(store.get(key) || '0', 10);
      const next = current - 1;
      store.set(key, String(next));
      return next;
    }),
    hget: vi.fn(async (key, field) => {
      const hash = store.get(key);
      return hash?.[field] || null;
    }),
    hset: vi.fn(async (key, field, value) => {
      let hash = store.get(key);
      if (!hash) {
        hash = {};
        store.set(key, hash);
      }
      hash[field] = value;
      return 1;
    }),
    hdel: vi.fn(async (key, field) => {
      const hash = store.get(key);
      if (!hash?.[field]) return 0;
      delete hash[field];
      return 1;
    }),
    lpush: vi.fn(async (key, ...values) => {
      let list = store.get(key);
      if (!list) {
        list = [];
        store.set(key, list);
      }
      list.unshift(...values);
      return list.length;
    }),
    rpush: vi.fn(async (key, ...values) => {
      let list = store.get(key);
      if (!list) {
        list = [];
        store.set(key, list);
      }
      list.push(...values);
      return list.length;
    }),
    lpop: vi.fn(async (key) => {
      const list = store.get(key);
      if (!list || list.length === 0) return null;
      return list.shift();
    }),
    rpop: vi.fn(async (key) => {
      const list = store.get(key);
      if (!list || list.length === 0) return null;
      return list.pop();
    }),
    sadd: vi.fn(async (key, ...members) => {
      let set = store.get(key);
      if (!set) {
        set = new Set();
        store.set(key, set);
      }
      let added = 0;
      for (const member of members) {
        if (!set.has(member)) {
          set.add(member);
          added++;
        }
      }
      return added;
    }),
    smembers: vi.fn(async (key) => {
      const set = store.get(key);
      return set ? Array.from(set) : [];
    }),
    sismember: vi.fn(async (key, member) => {
      const set = store.get(key);
      return set?.has(member) ? 1 : 0;
    }),
    flushall: vi.fn(async () => {
      store.clear();
      return 'OK';
    }),
    quit: vi.fn(async () => 'OK'),
    disconnect: vi.fn(),
    on: vi.fn(),
    status: 'ready'
  };
  
  return mockRedis;
}

/**
 * tests/helpers/apiCall.js — хелпер для API-запросов в тестах
 */

const BASE_URL = process.env.API_URL || 'http://localhost:3002';

export async function apiCall(endpoint, options = {}) {
  const {
    method = 'GET',
    body,
    token,
    cookies,
    headers = {}
  } = options;
  
  const url = endpoint.startsWith('http') ? endpoint : `${BASE_URL}${endpoint}`;
  
  const fetchOptions = {
    method,
    headers: {
      'Content-Type': 'application/json',
      ...headers
    }
  };
  
  if (token) {
    fetchOptions.headers['Authorization'] = `Bearer ${token}`;
  }
  
  if (cookies) {
    fetchOptions.headers['Cookie'] = cookies;
  }
  
  if (body && method !== 'GET') {
    fetchOptions.body = JSON.stringify(body);
  }
  
  const response = await fetch(url, fetchOptions);
  
  let data;
  const contentType = response.headers.get('content-type');
  if (contentType?.includes('application/json')) {
    data = await response.json();
  } else {
    data = await response.text();
  }
  
  return {
    status: response.status,
    statusText: response.statusText,
    headers: response.headers,
    data,
    ok: response.ok
  };
}

export async function loginViaApi(email, password) {
  const response = await apiCall('/api/auth/login', {
    method: 'POST',
    body: { email, password }
  });
  
  if (!response.ok) {
    throw new Error(`Login failed: ${response.status} ${JSON.stringify(response.data)}`);
  }
  
  // Извлекаем токен из cookie или body
  const setCookie = response.headers.get('set-cookie');
  const token = setCookie?.match(/sm_token=([^;]+)/)?.[1] || response.data.token;
  
  return {
    token,
    user: response.data.user,
    cookies: setCookie
  };
}

export async function healthCheck() {
  const response = await apiCall('/health/live');
  return response.ok;
}

export async function waitFor(condition, timeout = 5000, interval = 100) {
  const start = Date.now();
  while (Date.now() - start < timeout) {
    if (await condition()) return true;
    await new Promise(resolve => setTimeout(resolve, interval));
  }
  throw new Error('Timeout waiting for condition');
}
