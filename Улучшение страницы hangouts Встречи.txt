// src/lib/hangouts.ts
export function groupHangoutsByDate(hangouts: Hangout[]): [string, Hangout[]][] {
  const map = new Map<string, Hangout[]>();
  for (const h of hangouts) {
    const key = formatDateKey(h.event_date); // "today" | "tomorrow" | "2026-09-01"
    if (!map.has(key)) map.set(key, []);
    map.get(key)!.push(h);
  }
  return Array.from(map.entries());
}

export function formatDateGroupLabel(key: string, t: (k: string) => string): string {
  if (key === "today") return t("hangout.group.today");
  if (key === "tomorrow") return t("hangout.group.tomorrow");
  return new Date(key).toLocaleDateString("ru-RU", { weekday: "long", day: "numeric", month: "long" });
}

Проблема: кнопка «Показать ещё» требует точного попадания пальцем.
Решение: авто-подгрузка при приближении к низу + небольшой sentinel.

tsx
const sentinelRef = useRef<HTMLDivElement | null>(null);

useEffect(() => {
  if (!sentinelRef.current || !hasMore || loading) return;
  const observer = new IntersectionObserver(
    ([entry]) => { if (entry.isIntersecting) fetchNextPage(); },
    { rootMargin: "200px" }
  );
  observer.observe(sentinelRef.current);
  return () => observer.disconnect();
}, [hasMore, loading, fetchNextPage]);

  if (dateFilter !== "all") params.set("date", dateFilter);
  if (radius > 0) params.set("radius", String(radius));
  setSearchParams(params, { replace: true });
}, [debouncedSearch, category, dateFilter, radius, setSearchParams]);

// при маунте — читаем:
useEffect(() => {
  setSearch(searchParams.get("q") || "");
  setCategory(searchParams.get("category") || "all");
  setDateFilter(searchParams.get("date") || "all");
  setRadius(Number(searchParams.get("radius") || 0));
}, [searchParams]);
5. Pull-to-refresh (мобильный паттерн)
Проблема: на мобильном нет нативного жеста обновления.
Решение: лёгкая реализация на touch-событиях + интеграция с TanStack Query refetch.
Файл: src/pages/hangouts.tsx
tsx
const containerRef = useRef<HTMLDivElement>(null);
const [pullY, setPullY] = useState(0);
const startY = useRef(0);
const isPulling = useRef(false);

const onTouchStart = (e: React.TouchEvent) => {
  if (containerRef.current && containerRef.current.scrollTop === 0) {
    startY.current = e.touches[0].clientY;
    isPulling.current = true;
  }
};

const onTouchMove = (e: React.TouchEvent) => {
  if (!isPulling.current) return;
  const diff = e.touches[0].clientY - startY.current;
  if (diff > 0 && diff < 120) setPullY(diff * 0.5);
};

const onTouchEnd = () => {
  if (pullY > 60) {
    refetch(); // TanStack Query
  }
  isPulling.current = false;
  setPullY(0);
};

// в JSX:
<div
  ref={containerRef}
  onTouchStart={onTouchStart}
  onTouchMove={onTouchMove}
  onTouchEnd={onTouchEnd}
  style={{ transform: `translateY(${pullY}px)`, transition: isPulling.current ? "none" : "transform 0.2s" }}
>
  {pullY > 20 && (
    <div className="flex justify-center py-2 text-muted-foreground text-sm">
      {pullY > 60 ? t("hangout.pull.release") : t("hangout.pull.pull")}
    </div>
  )}
  {/* список */}
</div>
i18n: hangout.pull.pull → «Потяните чтобы обновить», hangout.pull.release → «Отпустите для обновления».
